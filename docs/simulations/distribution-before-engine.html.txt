function fitnessClamp(n){return Math.max(0,Math.min(1,n))}
function handFitnessBreakdown(state=G){
  // Only these three asset collections are read. No draws, commitments, temporary
  // bonuses, tracks, phase, distance, game flags, RNG or crisis definitions.
  const systems=state?.systems||[],crew=state?.crew||[],components=state?.components||{};
  let systemSum=0,systemWeight=0;
  for(const def of SYSTEM_DEFS){
    const weight=FITNESS_CONFIG.systems[def.id]||1;
    const health=systems.filter(s=>s.def.id===def.id).map(s=>s.state==="op"?1:s.state==="damaged"?FITNESS_CONFIG.damagedSystem:0).sort((a,b)=>b-a);
    const modular=MULTI_SYSTEMS.includes(def.id),target=modular?1+FITNESS_CONFIG.backupWeight:1;
    const effective=(health[0]||0)+(modular?FITNESS_CONFIG.backupWeight*(health[1]||0):0);
    systemSum+=weight*fitnessClamp(effective/target);systemWeight+=weight;
  }
  let crewSum=0,crewWeight=0;
  for(const c of crew){
    const weight=FITNESS_CONFIG.crew[c.id]||1;
    crewSum+=weight*(c.state==="ok"?1:c.state==="injured"?FITNESS_CONFIG.injuredCrew:0);crewWeight+=weight;
  }
  let componentSum=0,componentWeight=0;
  for(const type of COMP_TYPES){
    const weight=FITNESS_CONFIG.components[type];
    const quantity=Math.max(0,Number(components[type])||0);
    componentSum+=weight*fitnessClamp(quantity/FITNESS_CONFIG.componentsPerType);componentWeight+=weight;
  }
  const parts={systems:systemWeight?systemSum/systemWeight:0,crew:crewWeight?crewSum/crewWeight:0,components:componentWeight?componentSum/componentWeight:0};
  const viable=systems.some(s=>s.state==="op"||s.state==="damaged")&&crew.some(c=>c.state==="ok"||c.state==="injured");
  const value=viable?fitnessClamp(Object.entries(FITNESS_CONFIG.weights).reduce((sum,[key,weight])=>sum+weight*parts[key],0)):0;
  return {...parts,viable,value};
}

function addHeat(n){
  if(n===0)return;
  G.heat=Math.max(0,G.heat+n);
  if(n>0)log("Heat "+(n>0?"+":"")+n+" → "+G.heat+(G.heat>=5?" — CRITICAL":""),n>0?"bad":"good");
  else log("Heat "+n+" → "+G.heat,"good");
  if(G.heat>=5)scram();
}
function scram(){
  log("THERMAL LIMIT EXCEEDED — reactor SCRAM!","bad");
  let target=null;
  for(const id of POWER_SYSTEMS){const s=systemOfType(id);if(s.state!=="destroyed"){target=s;break}}
  if(target){damageStep(target,"reactor SCRAM / thermal overload");log(systemName(target)+" is "+(target.state==="destroyed"?"DESTROYED":"damaged")+" in the scram.","bad")}
  G.heat=3;G.flags.noRedline=true;
  log("Heat falls to 3. Power systems cannot be redlined next crisis.","bad");
}

function addCourse(n){G.course=Math.max(0,G.course+n);log("Course error "+(n>0?"+":"")+n+" → "+G.course,n>0?"bad":"good");if(G.course>=3)gameOver(false,"CONTACT LOST — TRAJECTORY MISSED","The ship slides past the corridor home. The signal fades. Nothing answers.")}

function deathDescription(c,cause){
  if(cause==="suffocation"){
    const ls=systemOfType("lifesupport"),penalties=G.ongoing;
    const mechanism=penalties.some(o=>o.key==="co2")?"Died from combined CO₂ poisoning and oxygen starvation as the cabin air became unbreathable.":"Died of hypoxic respiratory failure when the cabin's remaining oxygen was exhausted.";
    const support=ls.state==="destroyed"?" Every Life Support unit had been destroyed.":ls.state==="damaged"?" The surviving Life Support equipment was damaged and could not maintain the air supply.":"";
    const ongoing=penalties.length?" Active air-system failures: "+penalties.map(o=>o.text).join("; ")+".":"";
    return mechanism+support+ongoing;
  }
  if(cause!=="crisis consequence")return "Died from "+cause+" while already weakened by an unhealed injury.";
  const C=G.crisis,key=C?.id||Object.values(CRISES).flat().concat(REENTRY).find(x=>x.title===C?.title)?.id;
  let text=DEATH_DESCRIPTIONS[key];
  if(!text){
    const f=C?.fail||{};
    text=f.heat>0?"Died from heatstroke as the failed response allowed the compartment temperature to rise beyond survivable limits.":f.oxygen<0||f.ongoing?"Died from respiratory failure when the failed response left the cabin air unbreathable.":f.damageRandom||f.destroyRandom?"Died from traumatic internal injuries when damaged equipment broke loose during the failed response.":f.course>0?"Died from impact injuries during an uncontrolled attitude correction after the failed navigation response.":"Died from complications of further injuries sustained during the unsuccessful emergency response.";
  }
  return text+(c.injury?" Their earlier, unhealed injuries left them unable to survive the additional trauma.":"");
}
function incident(cause){return {cause,turn:G.turn,phase:G.phase,crisis:cause==="suffocation"?null:G.crisis?.title||null}}
function injureCrew(n,cause){
  for(let i=0;i<n;i++){
    if(G.flags.shelter>0){G.flags.shelter--;log("Emergency protection absorbs the harm ("+cause+").","sys");continue}
    const alive=G.crew.filter(c=>c.state!=="dead");if(!alive.length)break;
    const healthy=alive.filter(c=>c.state==="ok");
    const c=pick(healthy.length?healthy:alive);
    if(c.state==="ok"){c.state="injured";c.injury=incident(cause);log(c.name+" ("+c.role+") is injured — "+cause+". Ability lost.","bad")}
    else{c.state="dead";c.death={...incident(cause),description:deathDescription(c,cause),priorInjury:c.injury||null};log(c.name+" ("+c.role+") is dead — "+cause+".","bad")}
  }
  if(G.crew.every(c=>c.state==="dead"))gameOver(false,"CONTACT LOST — CREW LOST","The ship flies on, automated and empty. No one is left to bring home.");
}
function healCrew(n){
  for(let i=0;i<n;i++){const c=G.crew.find(x=>x.state==="injured");if(!c)return;c.state="ok";c.injury=null;log(c.name+" ("+c.role+") is patched up and back on duty.","good")}
}

// ---------- System state ----------
function damageStep(s,cause="crisis damage"){
  if(s.state==="op"){s.state="damaged";s.damage=incident(cause)}
  else if(s.state==="damaged")destroySys(s,cause);
}
function destroySys(s,cause="crisis destruction"){
  if(s.state==="destroyed")return;
  s.destruction=incident(cause);
  s.state="destroyed";s.mode=null;s.patched=false;G.drawn=G.drawn.filter(x=>x!==s.id);
  if(s.def.id==="cargo"&&!G.flags.dataLost){G.flags.dataLost=true;log("Mission data lost with "+systemName(s)+". A replacement cannot recover it.","bad")}
}
function damageRandomSystems(n){
  for(let i=0;i<n;i++){const pool=G.systems.filter(s=>s.state!=="destroyed");if(!pool.length)return;const s=pick(pool);damageStep(s,"random system damage");log(systemName(s)+" is "+(s.state==="destroyed"?"DESTROYED":"damaged")+".","bad")}
  if(G.systems.every(s=>s.state==="destroyed"))gameOver(false,"SHIP REDUCED TO A HULK","Every system is gone. The hull drifts, dark and silent.");
}
function destroyRandomSystems(n){
  for(let i=0;i<n;i++){const pool=G.systems.filter(s=>s.state!=="destroyed");if(!pool.length)return;const s=pick(pool);destroySys(s,"random system destruction");log(systemName(s)+" is DESTROYED.","bad")}
}
function damageEngine(){
  const pool=G.systems.filter(s=>ENGINE_SYSTEMS.includes(s.def.id)&&s.state!=="destroyed");
  if(pool.length){const s=pick(pool);damageStep(s,"failed trajectory burn");log(systemName(s)+" is "+(s.state==="destroyed"?"DESTROYED":"damaged")+" in the failed burn.","bad")}
  else damageRandomSystems(1);
}

  for(const t of spec)G.components[t]++;
}
function gainRandomComponents(n){for(let i=0;i<n;i++)G.components[pick(COMP_TYPES)]++}
function loseComponents(n){

}

function applyEffects(e,when){
  if(!e)return;
  if(e.heat)addHeat(e.heat);
  if(e.oxygen)addOxygen(e.oxygen);
  if(e.course)addCourse(e.course);
  if(e.injure)injureCrew(e.injure,"crisis consequence");
  if(e.heal)healCrew(e.heal);
  if(e.damageRandom)damageRandomSystems(e.damageRandom);
  if(e.destroyRandom)destroyRandomSystems(e.destroyRandom);
  if(e.damageSystem){const s=systemOfType(e.damageSystem);if(s.state!=="destroyed"){damageStep(s,"named system damage");log(systemName(s)+" is "+(s.state==="destroyed"?"DESTROYED":"damaged")+".","bad")}}
  if(e.damageSystem2){const s=systemOfType(e.damageSystem2);if(s.state!=="destroyed"){damageStep(s,"named system damage");log(systemName(s)+" is "+(s.state==="destroyed"?"DESTROYED":"damaged")+".","bad")}}
  if(e.components>0){gainRandomComponents(e.components);log("Recovered "+e.components+" component(s).","good")}
  if(e.components<0){loseComponents(-e.components);log("Burned through "+(-e.components)+" component(s).","bad")}
  if(e.ongoing){G.ongoing.push({...e.ongoing});log("Ongoing penalty: "+e.ongoing.text,"bad")}
  if(e.clearOngoing&&G.ongoing.length){G.ongoing=[];log("All ongoing penalties cleared.","good")}
  if(e.forecastHidden){G.flags.forecastHidden+=e.forecastHidden;log("Sensors degraded — crisis forecast lost for "+e.forecastHidden+" turns.","bad")}
  if(e.repairSystem){const s=systemsOfType(e.repairSystem).find(s=>s.state==="damaged");if(s&&s.state==="damaged"){s.state="op";s.damage=null;log(systemName(s)+" patched back to operational.","good")}}
  if(e.dataLost){G.flags.dataLost=true;log("Mission data uplink failed — the science is lost.","bad")}
  if(e.noShip){G.flags.noShip=true;log("Recovery beacon silent — the ship will not be recovered intact.","bad")}
  if(e.shelter){G.flags.shelter+=e.shelter;log("Habitat converted to emergency shelter: the next "+e.shelter+" injuries are absorbed.","good")}
}

// ---------- Turn flow ----------
function consume(){
  G.turn++;
  let drain=0;
  const ls=systemOfType("lifesupport");
  if(ls.state==="destroyed")drain=2;else if(ls.state==="damaged")drain=1;
  for(const o of G.ongoing){drain+=(-o.oxygen||0)+(o.initialDrain||0);o.initialDrain=0;}
  if(drain>0){G.oxygen=Math.max(0,G.oxygen-drain);log("Consume: −"+drain+" O₂ → "+G.oxygen+(ls.state!=="op"?" (Life Support "+ls.state+")":""),"bad")}
  else log("Consume: Life Support holds the air at "+G.oxygen+".","sys");
  if(G.oxygen===0){log("THE AIR IS GONE.","bad");injureCrew(1,"suffocation")}
  for(const o of G.ongoing){o.turns--}
  const expired=G.ongoing.filter(o=>o.turns<=0);
  for(const o of expired)log("Penalty expired: "+o.text,"sys");
  G.ongoing=G.ongoing.filter(o=>o.turns>0);
  if(G.flags.forecastHidden>0)G.flags.forecastHidden--;
}

function drawFive(){
  const alive=shuffle(G.systems.filter(s=>s.state!=="destroyed"));
  G.drawn=alive.slice(0,5).map(s=>s.id);
  for(const s of G.systems){s.mode=null;s.patched=false;s.seBoost=false;s.fabricated=null;s.specialists=[]}
  G.crewBonus={};G.pendingCmdr=false;G.pendingSE=false;G.pendingSpecialist=null;G.missionKeep=[];G.flags.fabricatedThisResponse=false;
  for(const c of G.crew)c.used=false;
}

function beginCrisis(doConsume=true){
  if(G.over)return;
  if(!G.deck.length){
    if(G.phase<3){G.step="route";G.crisis=null;render();return}
    G.crisis=REENTRY;
    if(doConsume)consume();if(G.over)return;
    drawFive();G.step="respond";
    log("▶ FINAL CRISIS: RE-ENTRY. Everything ends here.","sys");render();return;
  }
  if(doConsume)consume();if(G.over)return;
  G.crisis=G.deck.shift();
  if(G.flags.revealedNext===G.crisis.id)G.flags.revealedNext=null;
  drawFive();G.step="respond";
  log("▶ Crisis: "+G.crisis.title+" ["+G.crisis.cat+"]","sys");
  render();
}

// ---------- Player actions ----------
function setMode(id,mode){
  if(G.step!=="respond"||!G.crisis)return;
  const s=sys(id);if(!s||!G.drawn.includes(id)||s.state==="destroyed"||s.mode==="b")return;
  if(!["n","r","s"].includes(mode))return;
  if(mode==="r"){
    if(s.state!=="op")return;
    if(G.flags.noRedline&&POWER_SYSTEMS.includes(s.def.id)){log("Scram lockout: power systems cannot be redlined this crisis.","bad");return}
  }
  if(mode==="s"&&s.def.s==null)return;
  if(mode==="n"&&s.def.r==null&&Object.keys(s.def.n).length===0)return;
  s.mode=(s.mode===mode)?null:mode;
  // A mode button is also a target click: finish a pending matching specialist
  // assignment after committing, without requiring a second click on the card body.
  if(specialistCanTarget(crew(G.pendingSpecialist),s)){cardClick(id);return;}
  render();
}
function fabricationReason(builderId,type){
  if(G.over||G.step!=="respond"||!G.crisis)return "Fabrication is only available during a crisis response.";
  const builder=sys(builderId),def=SYSTEM_DEFS.find(d=>d.id===type);
  if(!builder||builder.def.id!=="fabricator"||!G.drawn.includes(builder.id))return "Draw the Fabricator to fabricate.";
  if(builder.state!=="op")return "Repair the Fabricator to fabricate; a patch is not enough.";
  if(builder.mode)return "The Fabricator is already committed this crisis.";
  if(!def)return "Unknown system blueprint.";
  if(!MULTI_SYSTEMS.includes(type)&&systemsOfType(type).some(s=>s.state!=="destroyed"))return "Only one surviving "+def.name+" is allowed; destroy the existing one before building a replacement.";
  if(G.components[def.comp]<FABRICATION_COST)return "Needs "+FABRICATION_COST+"× "+def.comp+" (you have "+G.components[def.comp]+").";
  return "";
}
function fabricate(builderId,type){
  const reason=fabricationReason(builderId,type);
  if(reason){log(reason,"bad");return false;}
  const builder=sys(builderId),def=SYSTEM_DEFS.find(d=>d.id===type);
  const serial=systemsOfType(type).length+1;
  const created=makeSystem(def,type+"#"+serial);
  G.components[def.comp]-=FABRICATION_COST;
  G.systems.push(created);
  builder.mode="b";builder.fabricated=created.id;G.flags.fabricatedThisResponse=true;
  log("Fabricated "+systemName(created)+" (−"+FABRICATION_COST+" "+def.comp+"). Operational; enters future system draws. "+systemName(builder)+" produces no capabilities this crisis.","good");
  closeModal();render();return created;
}
function cannibaliseGains(s){
  const gains=s.def.yield3?[s.def.comp,s.def.comp,s.def.comp]:[s.def.comp,s.def.comp,"Wiring"];
  if(crew("eng")?.state==="ok")gains.push(s.def.comp);
  return gains;
}
function cannibalise(id){
  if(G.step!=="respond"||G.over)return false;
  const s=sys(id);if(!s||s.state!=="op")return false;
  destroySys(s,"cannibalised for components");
  const gains=cannibaliseGains(s);
  gainComponents(gains);
  log("Cannibalised "+systemName(s)+" → "+gains.join(", ")+". It will never fly again.","bad");
  render();
}
function repair(id){
  if(G.step!=="respond")return;
  const s=sys(id);if(s.state!=="damaged")return;
  if(G.components[s.def.comp]<2){log("Repair needs 2× "+s.def.comp+".","bad");return}
  G.components[s.def.comp]-=2;s.state="op";s.damage=null;s.patched=false;
  log("Repaired "+systemName(s)+" to operational (−2 "+s.def.comp+").","good");render();
}
function patch(id){
  if(G.step!=="respond")return;
  const s=sys(id);if(s.state!=="damaged"||s.patched)return;
  const t=COMP_TYPES.find(t=>G.components[t]>0);
  if(!t){log("Patch needs 1 component of any type.","bad");return}
  G.components[t]--;s.patched=true;
  log("Patched "+systemName(s)+" (−1 "+t+") — full Normal value for this crisis.","good");render();
}
function missionReason(){
  if(!G||G.over||G.step!=="respond"||!G.crisis)return "Mission abilities require a crisis response.";
  const c=crew("spec");if(!c||c.state!=="ok")return "The Mission Specialist must be healthy.";
  if(G.flags.specUsed||c.used)return "This Mission ability has already been spent.";
  if(c.mission==="kim"&&(G.flags.fabricatedThisResponse||G.systems.some(s=>s.mode)))return "Redraw before committing systems or fabricating.";
  if(c.mission==="adeyemi"&&!G.crew.some(x=>x.state==="injured"))return "No injured crew to heal.";
  if(c.mission==="voss"&&!G.ongoing.length&&G.oxygen===6)return "No ongoing penalty and Oxygen is already full.";
  if(c.mission==="nasser"&&G.course===0&&!G.deck.length)return "No Course Error or queued crisis to reveal.";
  if(c.mission==="chen"&&G.phase===3&&!G.deck.length)return "No unknown future crisis forecasts remain.";
  return "";
}
function redrawMissionSystems(keep){
  if(!Array.isArray(keep)||keep.length>2||new Set(keep).size!==keep.length||keep.some(id=>!G.drawn.includes(id)||sys(id)?.state==="destroyed"))return false;
  const old=G.drawn.slice(),alive=G.systems.filter(s=>s.state!=="destroyed"),size=Math.min(5,alive.length),need=size-keep.length;
  const fresh=shuffle(alive.filter(s=>!old.includes(s.id))).map(s=>s.id);
  const fallback=shuffle(alive.filter(s=>old.includes(s.id)&&!keep.includes(s.id))).map(s=>s.id);
  G.drawn=keep.concat(fresh.slice(0,need),fallback.slice(0,Math.max(0,need-fresh.length)));
  return true; // Do NOT reset crew uses, bonuses, health, patches, boosts or construction flags.
}

    case"ortega":G.flags.shelter+=2;log("Dr. Ortega establishes emergency procedures: the next 2 injury hits are absorbed.","good");break;

      if(!redrawMissionSystems(choice))return false;

// ---------- Totals ----------
function half(obj){const o={};for(const k in obj)o[k]=Math.floor(obj[k]/2);return o}
function addInto(t,obj){for(const k in obj)t[k]=(t[k]||0)+obj[k]}
function systemOutput(s,mode=s.mode,fullNormal=false){
  if(s.state==="destroyed"||!["n","r","s"].includes(mode))return {};
  const raw=s.def[mode];if(!raw)return {};
  const eff=mode==="n"&&s.state==="damaged"&&!s.patched&&!s.seBoost&&!fullNormal?half(raw):{...raw};
  for(const id of s.specialists){const c=crew(id);if(c?.state==="ok"&&(raw[c.cap]||0)>0)eff[c.cap]=(eff[c.cap]||0)+2;}
  return eff;
}
function computeTotals(){
  const t={};
  for(const id of G.drawn)addInto(t,systemOutput(sys(id)));
  addInto(t,G.crewBonus);
  return t;
}
function computeHeat(){
  let h=0;
  for(const id of G.drawn){const s=sys(id);if(!s.mode)continue;
    h+=s.mode==="n"?(s.def.nh||0):s.mode==="r"?(s.def.rh||0):s.mode==="s"?(s.def.sh||0):0}
  return h;
}

// ---------- Resolution ----------
function crisisRequirementsMet(c,totals){
  return [c.req,...(c.alt?[c.alt]:[])].some(req=>Object.entries(req).every(([k,v])=>(totals[k]||0)>=v));
}
function resolveCrisis(){
  if(G.step!=="respond"||!G.crisis)return;
  const C=G.crisis;
  G.flags.noRedline=false; // scram lockout only lasts one crisis
  G.pendingCmdr=false;G.pendingSE=false;G.pendingSpecialist=null;
  const totals=computeTotals();
  const arms=[C.req];if(C.alt)arms.push(C.alt);
  const success=crisisRequirementsMet(C,totals);
  // heat: generated minus surplus cooling (surplus measured against the cheapest arm)
  const clgMin=Math.min(...arms.map(a=>a.CLG||0));
  const surplus=Math.max(0,(totals.CLG||0)-clgMin);
  const heatNet=computeHeat()-surplus;
  if(heatNet!==0)addHeat(heatNet);
  if(surplus>0)log("Surplus cooling bleeds off "+surplus+" heat.","good");
  // sacrifice side-effects
  for(const id of G.drawn.slice()){
    const s=sys(id);
    if(s.mode==="s"&&s.def.sfx)applyEffects(s.def.sfx);
  }
  // surplus Navigation corrects drift: nav capability spent beyond the requirement pays down course error
  const navMin=Math.min(...arms.map(a=>a.NAV||0));
  const surplusNav=Math.max(0,(totals.NAV||0)-navMin);
  if(!C.route&&surplusNav>=3&&G.course>0){addCourse(-1);log("Surplus navigation capacity trims the course error.","good")}
  // outcome
  if(C.route){
    const target=C.target;
    if(success){log("Burn nominal — the short route holds. Two journey cards fall away.","good");buildDeck(target,BASE_DECK[target]-2)}
    else{log("The burn falls short. The engines pay for it.","bad");damageEngine();buildDeck(target,BASE_DECK[target]-1)}
    G.phase=target;G.flags.routeTarget=null;
    log("Phase "+["","I — ACCIDENT","II — TRANSIT","III — RETURN"][target]+". Distance home ≈ "+distanceEstimate()+" crises.","sys");
    G.crisis=null;beginCrisis(true);return;
  }
  if(success){
    log("RESOLVED: "+C.title+". The ship limps onward.","good");
    applyEffects(C.win);
  }else{
    const missing=arms.map(req=>Object.entries(req).filter(([k,v])=>(totals[k]||0)<v).map(([k,v])=>CAPS[k]+" "+(totals[k]||0)+"/"+v).join(", ")).join("  — or —  ");
    log("FAILED: "+C.title+" ("+missing+").","bad");
    applyEffects(C.fail);
  }
  if(G.over){render();return;}
  // mode consequences
  for(const id of G.drawn.slice()){
    const s=sys(id);
    if(s.mode==="r"){damageStep(s,"redlining");log(systemName(s)+" is "+(s.state==="destroyed"?"DESTROYED":"damaged")+" by redlining.","bad")}
    else if(s.mode==="s"){destroySys(s,"sacrificed for crisis output");log(systemName(s)+" is sacrificed — gone for good.","bad")}
    s.patched=false;s.mode=null;s.seBoost=false;
  }
  if(G.over)return;
  if(C.terminal){endGame(success);return}
  G.crisis=null;
  beginCrisis(true);
}

function buildDeck(phase,count){
  const n=Math.max(2,Math.min(count,G.pools[phase].length));
  G.deck=G.pools[phase].slice(0,n);

function distanceEstimate(){
  let d=G.deck.length+(G.crisis&&!G.crisis.route?1:0);
  if(G.phase<2)d+=BASE_DECK[2];
  if(G.phase<3)d+=BASE_DECK[3];
  if(!(G.crisis&&G.crisis.terminal))d+=1; // re-entry
  return d;
}

function cardValsText(s){
  const d=s.def,lines=[];

function postMortem(){
  const repairs=Object.fromEntries(COMP_TYPES.map(t=>[t,0])),replacements={...repairs};
  const systems=G.systems.map(s=>{
    const cost=s.state==="damaged"?2:s.state==="destroyed"?FABRICATION_COST:0;
    if(s.state==="damaged")repairs[s.def.comp]+=cost;
    if(s.state==="destroyed")replacements[s.def.comp]+=cost;

}

document.addEventListener("click",e=>{

  const act=b.dataset.act,id=b.dataset.id;
  switch(act){
    case"mode":setMode(id,b.dataset.mode);break;
    case"cann":cannibalise(id);break;
    case"repair":repair(id);break;
    case"patch":patch(id);break;
    case"crew":crewClick(id);break;
    case"req":reqClick(id);break;
    case"card":cardClick(id);break;
    case"resolve":resolveCrisis();break;
    case"route-safe":routeSafe();break;
    case"route-direct":routeDirect();break;
    case"help":showHelp();break;
    case"postmortem":showGameOverModal();break;
    case"fabrication":showFabrication(id);break;
    case"fabricate":fabricate(id,b.dataset.type);break;
    case"mission-effect":useMission(b.dataset.choice);break;
    case"mission-keep":toggleMissionKeep(id);break;
    case"mission-redraw":useMission(G.missionKeep.slice());break;
    case"closemodal":closeModal();break;
    case"difficulty-start":if(Object.hasOwn(DIFFICULTIES,b.dataset.level))location.search=runQuery(SEED,b.dataset.level);break;
    case"newgame":{const s=(Math.random()*1e9)|0;location.search=runQuery(s);break}
    case"restartgame":closeModal();newGame();break;
    case"debug":debugFitness=!debugFitness;renderFitness();break;
  }
});
